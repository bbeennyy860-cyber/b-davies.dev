signal_engine.py 
class SignalEngine:
    def __init__(self, feeds: list):
        self.feeds = feeds
        self.conviction = 0.0

    def score(self, token: str) -> float:
        signals = [
            f.evaluate(token)
            for f in self.feeds
        ]
        return sum(signals) / len(signals)
